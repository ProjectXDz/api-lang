// ==================== HELPERS ====================

// Fisher-Yates shuffle
function shuffle(arr) {
const a = […arr];
for (let i = a.length - 1; i > 0; i–) {
const j = Math.floor(Math.random() * (i + 1));
[a[i], a[j]] = [a[j], a[i]];
}
return a;
}

function filterWords(lang, level) {
let words = vocabulary[lang] || [];
if (level) words = words.filter((w) => w.level === Number(level));
return words;
}

// ==================== ROUTES ====================

// GET /api/languages
// รายชื่อภาษาทั้งหมดที่มี
app.get(”/api/languages”, (req, res) => {
const langs = Object.keys(vocabulary).map((lang) => ({
language: lang,
total: vocabulary[lang].length,
levels: […new Set(vocabulary[lang].map((w) => w.level))].sort(),
}));
res.json({ languages: langs });
});

// GET /api/words?lang=japanese&level=2&limit=5
// ดึงคำศัพท์ตาม lang และ level พร้อม pagination
app.get(”/api/words”, (req, res) => {
const { lang = “japanese”, level, offset = 0, limit = 10 } = req.query;

if (!vocabulary[lang]) {
return res.status(404).json({ error: `Language "${lang}" not found` });
}

const words = filterWords(lang, level);
const paginated = words.slice(Number(offset), Number(offset) + Number(limit));

res.json({
language: lang,
level: level ? Number(level) : “all”,
total: words.length,
offset: Number(offset),
limit: Number(limit),
words: paginated,
});
});

// GET /api/words/random?lang=japanese&level=3
// สุ่มคำ 1 คำ
app.get(”/api/words/random”, (req, res) => {
const { lang = “japanese”, level } = req.query;

if (!vocabulary[lang]) {
return res.status(404).json({ error: `Language "${lang}" not found` });
}

const words = filterWords(lang, level);
if (words.length === 0) {
return res.status(404).json({ error: “No words found for this filter” });
}

const word = words[Math.floor(Math.random() * words.length)];
res.json({ language: lang, word });
});

// POST /api/words/random/batch
// สุ่มคำหลายคำ โดยไม่ซ้ำกัน (no-repeat)
// Body: { lang, level, count, exclude: [“id1”,“id2”,…] }
app.post(”/api/words/random/batch”, (req, res) => {
const { lang = “japanese”, level, count = 5, exclude = [] } = req.body;

if (!vocabulary[lang]) {
return res.status(404).json({ error: `Language "${lang}" not found` });
}

let words = filterWords(lang, level);

// กรองคำที่เคยเห็นแล้วออก (no-repeat)
words = words.filter((w) => !exclude.includes(w.id));

if (words.length === 0) {
return res.json({
language: lang,
words: [],
exhausted: true,
message: “All words have been seen. Reset to start over.”,
});
}

const picked = shuffle(words).slice(0, Math.min(count, words.length));

res.json({
language: lang,
level: level ? Number(level) : “all”,
words: picked,
remaining: words.length - picked.length,
exhausted: false,
});
});
